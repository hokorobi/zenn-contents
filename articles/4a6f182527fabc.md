こちらを読んで試してみました。
とりあえず普通に使えるようになったようなので、本格移行するための設定を追加しようと思っています。

## 移行作業

こんな感じで移行しました。

```sh
mkdir ~/.cache/goda-pm/repos/github.com/Shougo
cd ~/.cache/goda-pm/repos/github.com/Shougo
git clone https://github.com/Shougo/goda-pm.vim
git clone https://github.com/Shougo/goda.vim
git clone https://github.com/Shougo/goda-pm-ext-lazy.vim
```

~/vimfiles/plug/conf に以下のファイルを追加。

- `goda-pm.vim`
- `goda-pm/main.go`
- `_goda.toml`

```vim:goda-pm.vim
vim9script

# goda-pm.vim must be in 'runtimepath'.
set runtimepath^=~/.cache/goda-pm/repos/github.com/Shougo/goda-pm.vim

const base_dir = '~/.cache/goda-pm'
$BASE_DIR = '<script>:h'->expand()
const g:goda_pm#config_dir = $'{$BASE_DIR}/goda-pm'

if base_dir->goda_pm#min#load_state()
  # No state: goda.vim and goda-pm-ext-lazy.vim are needed to make it.
  set runtimepath^=~/.cache/goda-pm/repos/github.com/Shougo/goda.vim
  set runtimepath^=~/.cache/goda-pm/repos/github.com/Shougo/goda-pm-ext-lazy.vim
  autocmd VimEnter * ++once call goda_pm#make_state(base_dir)
endif

autocmd User GodaPm:makeStatePost echomsg 'goda_pm#make_state() is done'
```

```go:goda-pm/main.go
package main

import (
	"context"
	"path/filepath"

	"github.com/Shougo/goda-pm-ext-lazy.vim/ext/lazy"
	"github.com/Shougo/goda-pm-ext/ext/installer"
	"github.com/Shougo/goda-pm-ext/ext/local"
	"github.com/Shougo/goda-pm-ext/ext/protocol/git"
	"github.com/Shougo/goda-pm-ext/ext/protocol/http"
	"github.com/Shougo/goda-pm-ext/ext/toml"
	"github.com/Shougo/goda-pm.vim/pm"
)

func main() {
	b := pm.New()
	b.AddProtocol(git.New())
	b.AddProtocol(http.New())
	b.AddExt(installer.New())
	b.SetConfig(config)
	b.Main()
}

func config(ctx context.Context, c *pm.ConfigArgs) (*pm.ConfigReturn, error) {

	c.Options.ExtParams["installer"] = pm.Params{
		"checkDiff":     true,
		"logFilePath":   "~/.cache/goda-pm/installer-log.txt",
		"minCommitDays": 1,
	}
	c.Options.Protocols = []string{"git", "http"}

	// The plugins by name: a later one replaces an earlier one.
	recordPlugins := map[string]pm.Plugin{}
	var names []string
	record := func(p pm.Plugin) {
		if _, ok := recordPlugins[p.Name()]; !ok {
			names = append(names, p.Name())
		}
		recordPlugins[p.Name()] = p
	}
	ftplugins := map[string]string{}
	var hooksFiles []string
	var multipleHooks []pm.MultipleHook

	for _, f := range []struct {
		path string
	}{
		{
			"$BASE_DIR/_plugins.toml",
		},
		{
			"$BASE_DIR/_goda.toml",
		},
		{
			"$BASE_DIR/_denops.toml",
		},
		{
			"$BASE_DIR/_reference.toml",
		},
		{
			"$BASE_DIR/_builtin.toml",
		},
	} {
		t, err := toml.Load(ctx, c.Host, f.path, nil)
		if err != nil {
			return nil, err
		}
		for _, p := range t.Plugins {
			record(p)
		}
		pm.MergeFtplugins(ftplugins, t.Ftplugins)
		multipleHooks = append(multipleHooks, t.MultipleHooks...)
		if t.HooksFile != "" {
			hooksFiles = append(hooksFiles, t.HooksFile)
		}
	}

	// The plugins in ~/work are used from there, instead of those of the
	// TOML files of the same name.
	localPlugins, err := local.Load(
		ctx,
		c.Host,
		"~/work",
		pm.Plugin{"frozen": true, "merged": false},
		[]string{"*.nvim", "*.vim", "nvim-*", "vim*"},
	)
	if err != nil {
		return nil, err
	}
	for _, p := range localPlugins {
		if old, ok := recordPlugins[p.Name()]; ok {
			url, err := c.URL(ctx, "git", old)
			if err != nil {
				return nil, err
			}
			p = pm.Merge(old, p, pm.Plugin{"url": url})
		}
		record(p)
	}

	plugins := make([]pm.Plugin, 0, len(names))
	for _, n := range names {
		plugins = append(plugins, recordPlugins[n])
	}
	lazyResult, err := lazy.MakeState(ctx, c.Host, plugins)
	if err != nil {
		return nil, err
	}

	baseDir, err := c.Expand(ctx, "$BASE_DIR")
	if err != nil {
		return nil, err
	}
	checkFiles, err := filepath.Glob(filepath.Join(baseDir, "*"))
	if err != nil {
		return nil, err
	}

	return &pm.ConfigReturn{
		CheckFiles:    checkFiles,
		Ftplugins:     ftplugins,
		HooksFiles:    hooksFiles,
		MultipleHooks: multipleHooks,
		Plugins:       lazyResult.Plugins,
		StateLines:    lazyResult.StateLines,
	}, nil
}
```

```toml:_goda.toml
[[plugins]]
repo = 'https://github.com/Shougo/goda.vim'
merged = false

[[plugins]]
repo = 'https://github.com/Shougo/goda-pm.vim'
hook_post_update = 'call goda_pm#build()'
merged = false

[[plugins]]
repo = 'Shougo/goda-pm-ext-lazy.vim'
merged = false
```

```sh
cd ~/vimfiles/plug/conf/goda-pm
go mod init goda-pm-config
go get github.com/Shougo/goda-pm-ext@latest
go get github.com/Shougo/goda-pm-ext-lazy.vim@latest
go mod tidy
```

vimrc で dpp.vim を使うために読み込んでいる dpp.vim を goda-pm.vim を読み込むように変更
```vim
execute $'source {expand("<script>:h")}/plug/conf/goda-pm.vim'
```

Vim を再起動して ```:call goda_pm#async_ext_action('installer', 'install')```
